window.exportExcel = async function () { const exportButton = document.querySelector('.btn-secondary'); try { if (exportButton) { exportButton.disabled = true; exportButton.innerHTML = '⏳ Mengambil Data...'; } Swal.fire({ title: 'Menyiapkan Data', text: 'Sedang mengambil seluruh data presensi...', allowOutsideClick: false, didOpen: () => { Swal.showLoading(); } }); // Ambil seluruh data dengan pagination // Supabase biasanya membatasi jumlah row per request const pageSize = 1000; let from = 0; let seluruhData = []; while (true) { const { data, error } = await supabaseClient .from('presensi') .select('*') .order('created_at', { ascending: false }) .range(from, from + pageSize - 1); if (error) { throw error; } if (!data || data.length === 0) { break; } seluruhData = seluruhData.concat(data); // Jika hasil kurang dari 1000 berarti sudah halaman terakhir if (data.length < pageSize) { break; } from += pageSize; } if (!seluruhData.length) { Swal.fire({ icon: 'info', title: 'Data Kosong', text: 'Belum ada data presensi untuk diexport.' }); return; } // Format data untuk Excel const exportData = seluruhData.map((item, index) => { const waktu = item.created_at ? new Date(item.created_at) : null; return { No: index + 1, Tanggal: waktu ? waktu.toLocaleDateString('id-ID') : '-', Waktu: waktu ? waktu.toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit', second: '2-digit' }) : '-', Nama: item.nama || '-', NIK: item.nik || '-', Jabatan: item.jabatan || '-', Departemen: formatDept(item.departemen), Status: item.status || '-', Keterangan: item.keterangan || '-' }; }); // Buat worksheet const ws = XLSX.utils.json_to_sheet(exportData); // Atur lebar kolom ws['!cols'] = [ { wch: 6 }, // No { wch: 14 }, // Tanggal { wch: 12 }, // Waktu { wch: 28 }, // Nama { wch: 18 }, // NIK { wch: 25 }, // Jabatan { wch: 30 }, // Departemen { wch: 14 }, // Status { wch: 35 } // Keterangan ]; // Buat workbook const wb = XLSX.utils.book_new(); XLSX.utils.book_append_sheet( wb, ws, 'Seluruh Presensi' ); // Tanggal export untuk nama file const sekarang = new Date(); const tanggalFile = sekarang.getFullYear() + '-' + String(sekarang.getMonth() + 1).padStart(2, '0') + '-' + String(sekarang.getDate()).padStart(2, '0'); // Download Excel XLSX.writeFile( wb, `Rekap-Seluruh-Presensi-${tanggalFile}.xlsx` ); Swal.fire({ icon: 'success', title: 'Export Berhasil', text: `${seluruhData.length} data presensi berhasil diexport.` }); } catch (error) { console.error('Export error:', error); Swal.fire({ icon: 'error', title: 'Export Gagal', text: error.message || 'Terjadi kesalahan saat mengambil data.' }); } finally { if (exportButton) { exportButton.disabled = false; exportButton.innerHTML = '⬇ Export Semua Data'; } } };
